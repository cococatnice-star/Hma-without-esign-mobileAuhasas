local function isCodeActivationMessage(text)
    if type(text) ~= "string" then return false end
    local clean = text:lower():gsub("<[^>]->", " "):gsub("[%c]", " "):gsub("%s+", " ")
    return clean:find("%f[%a]code+%s+is%f[%A]") ~= nil
        or clean:find("%f[%a]the%s+code+%f[%A]") ~= nil
        or clean:find("%f[%a]use%s+code+%f[%A]") ~= nil
end

local function clickButton(btn)
    if not btn then return false end
    if typeof(firesignal) == "function" then
        local ok = pcall(function() firesignal(btn.Activated) end)
        if ok then return true end
    end
    if typeof(getconns) == "function" then
        local ok, connections = pcall(getconns, btn.Activated)
        if ok and type(connections) == "table" then
            for _, connection in ipairs(connections) do
                local fired = pcall(function() connection:Fire() end)
                if fired then return true end
            end
        end
    end
    if typeof(fireclick) == "function" then
        return pcall(function() fireclick(btn) end)
    end
    return false
end
local function fireBoxFocusLost(box)
    if not box then return false end
    local anyFired = false
    if typeof(firesignal) == "function" then
        local ok = pcall(firesignal, box.FocusLost, true)
        anyFired = anyFired or ok
    end
    if typeof(getconns) == "function" then
        local ok, cs = pcall(getconns, box.FocusLost)
        if ok and type(cs) == "table" then
            for _, c in ipairs(cs) do
                local fn
                pcall(function() fn = c.Function end)
                if fn and typeof(getupvalues) == "function" and typeof(setupv) == "function" then
                    local uOk, ups = pcall(getupvalues, fn)
                    if uOk and type(ups) == "table" then
                        for i, v in pairs(ups) do
                            if type(v) == "boolean" and v == true then
                                pcall(setupv, fn, i, false)
                            end
                        end
                    end
                end
                local fOk = pcall(function()
                    if c.Enabled ~= false then c:Fire(true) end
                end)
                anyFired = anyFired or fOk
            end
        end
    end
    return anyFired
end

local function aceStripRich(text)
    if type(text) ~= "string" then return tostring(text) end
    return (text:gsub("<[^>]->", ""))
end
local function aceExtractText(...)
    local args = table.pack(...)
    local keys = {"Text", "text", "Message", "message", "Content", "content", "Title", "title"}
    for index = 1, args.n do
        local value = args[index]
        if type(value) == "string" and value:match("%S") then
            return aceStripRich(value):match("^%s*(.-)%s*$") or ""
        end
        if type(value) == "table" then
            for _, key in ipairs(keys) do
                local field = value[key]
                if type(field) == "string" and field:match("%S") then
                    return aceStripRich(field):match("^%s*(.-)%s*$") or ""
                end
            end
        end
    end
    return ""
end
local function aceTokenize(text)
    local words = {}
    for word in text:gmatch("[%w_]+") do
        words[#words + 1] = word
    end
    return words
end

end

local riddleAnswers = {
					["what is the best strategy for stealing"] = "Speed Boots plus scouting weakly defended, recently unlocked bases",

local function normalizeRiddleText(text)
    if type(text) ~= "string" then return "" end
    return text:lower():gsub("<[^>]+>", " "):gsub("[%p%s]+", " "):gsub("^%s+", ""):gsub("%s+$", "")
end

local function looksLikeRiddleQuestion(text)
    if type(text) ~= "string" then return false end
    local clean = text:gsub("<[^>]+>", " "):gsub("^%s+", ""):gsub("%s+$", "")
    if #clean < 3 or isCodeActivationMessage(clean) then return false end
    local lower = clean:lower()
    if lower:find("sammy", 1, true) then return true end
    if clean:find("?", 1, true) then return true end
    local first = normalizeRiddleText(clean):match("^(%S+)")
    local starts = {
        ["what"] = true, ["who"] = true, ["where"] = true, ["when"] = true, ["why"] = true, ["how"] = true,
        ["which"] = true, ["whose"] = true, ["can"] = true, ["do"] = true, ["does"] = true, ["did"] = true,
        ["is"] = true, ["are"] = true, ["was"] = true, ["were"] = true, ["will"] = true, ["would"] = true,
        ["could"] = true, ["should"] = true, ["solve"] = true, ["guess"] = true
    }
    if starts[first or ""] then return true end
    for _, word in ipairs({"favorite","best","worst","rarest","oldest","newest","first","last","brainrot","mutation","trait","secret","machine","developer","studio"}) do
        if lower:find(word, 1, true) then return true end
    end
    return false
end

local function findRiddleAnswer(question)
    local normalized = normalizeRiddleText(question)
    if normalized == "" then return nil end
    local exact = riddleAnswers[normalized]
    if exact then return tostring(exact) end
    local qwords = {}
    for word in normalized:gmatch("%S+") do
        if #word > 2 then qwords[word] = true end
    end
    local bestAnswer = nil
    local bestScore = 0
    local tied = false
    for key, answer in pairs(riddleAnswers) do
        if type(key) == "string" then
            local keyWords = {}
            for word in normalizeRiddleText(key):gmatch("%S+") do
                if #word > 2 then keyWords[#keyWords + 1] = word end
            end
            if #keyWords >= 2 then
                local hits = 0
                for _, word in ipairs(keyWords) do
                    if qwords[word] then hits = hits + 1 end
                end
                local score = hits / #keyWords
                if hits >= 2 and score >= 0.9 then
                    if score > bestScore then
                        bestScore = score
                        bestAnswer = tostring(answer)
                        tied = false
                    elseif score == bestScore and tostring(answer) ~= bestAnswer then
                        tied = true
                    end
                end
            end
        end
    end
    if tied then return nil end
    return bestAnswer
end

end

local function solveRiddleQuestion(question)
    if not _riddleEnabled or not looksLikeRiddleQuestion(question) then return false end
    local now = os.clock()
    local normalized = normalizeRiddleText(question)
    if normalized == normalizeRiddleText(_lastRiddleQuestion) and now - _lastRiddleAt < 2 then return false end
    local answer = novaWebRiddleAnswer(question)
    if not answer or answer == "" or answer == "Unknown" then
        answer = findRiddleAnswer(question)
    end
    if not answer or answer == "" or answer == "Unknown" then return false end
    _lastRiddleAt = now
    updateRiddleDisplay(question, answer)
    if setStatus then setStatus("Riddle answer: " .. answer, COLORS.Green) end
    if _riddleAutoSubmit then
        task.spawn(function()
            local ok, msg = typeAndSubmitCode(string.upper(answer))
            if setStatus then
                setStatus(ok and ("Riddle redeemed: " .. answer) or ("Riddle redeem failed: " .. tostring(msg)), ok and COLORS.Green or COLORS.Red)
            end
        end)
    end
    return true
end

local function makeStateButton(parent, enabled, consoleName, onToggle)

    button.Name = "State"

    local function applyState(newState, notify)
        newState = newState == true

        state = newState
        featureStates[consoleName] = state

        TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {

            Transparency = state and 0.62 or 0.88,
        }):Play()
        if notify and autoWriteEnabled then appendConsoleStatus(consoleName, state) end
        if notify and onToggle then onToggle(state) end
    end
    featureStates.__setters[consoleName] = function(newState, notify)
        applyState(newState, notify ~= false)
    end

        if tick() - lastSubToggle < 0.15 then return end

        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            toggleState()
        end
    end)
    return button
end

    saveConfig()

    local token = autoBuyToken

            local character = LocalPlayer.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if root then
                local nearestPrompt, nearestDistance
                for _, object in ipairs(workspace:GetDescendants()) do
                    if object:IsA("ProximityPrompt") and object.Enabled then
                        local text = ((object.ActionText or "") .. " " .. (object.ObjectText or "") .. " " .. object.Name):lower()
                        if text:find("buy", 1, true) or text:find("purchase", 1, true) then
                            local parent = object.Parent
                            local position
                            if parent and parent:IsA("BasePart") then position = parent.Position
                            elseif parent and parent:IsA("Attachment") and parent.Parent and parent.Parent:IsA("BasePart") then position = parent.Parent.Position end
                            if position then
                                local distance = (root.Position - position).Magnitude
                                if distance <= (object.MaxActivationDistance or 10) + 2 and (not nearestDistance or distance < nearestDistance) then
                                    nearestPrompt, nearestDistance = object, distance
                                end
                            end
                        end
                    end
                end
                if nearestPrompt then
                    pcall(function()
                        if type(fireproximityprompt) == "function" then
                            fireproximityprompt(nearestPrompt)
                        else
                            nearestPrompt:InputHoldBegin()
                            task.wait(math.max(nearestPrompt.HoldDuration, 0.05) + 0.05)
                            nearestPrompt:InputHoldEnd()
                        end
                    end)
                end
            end
            task.wait(0.2)
        end
    end)
end

local function sendSpawnToDiscord(message)

        })
        local options = {
            Url = DISCORD_RELAY_URL,
            Method = "POST",
            Headers = {["Content-Type"] = "application/json"},
            Body = body,
        }
        local requestFunction = (syn and syn.request)
            or (http and http.request)
            or http_request
            or request
        if type(requestFunction) == "function" then
            pcall(requestFunction, options)
        else
            pcall(function()
                return HttpService:RequestAsync(options)
            end)
        end
    end)
end
local function extractSpawnedBrainrot(text)
    local clean = tostring(text or ""):gsub("<[^>]->", ""):gsub("[%c]", " "):gsub("%s+", " ")
    clean = clean:match("^%s*(.-)%s*$") or ""
    if clean == "" then return nil end
    local lower = clean:lower()
    local startAt = lower:find("brainrot spawned:", 1, true)
    local name
    if startAt then
        name = clean:sub(startAt + #"brainrot spawned:")
    else
        local phraseAt = lower:find(" has spawned", 1, true)
        if phraseAt then
            name = clean:sub(1, phraseAt - 1)
        else
            phraseAt = lower:find(" spawned", 1, true)
            if phraseAt then name = clean:sub(1, phraseAt - 1) end
        end
    end
    if not name and lower:find("brainrot%s+spawned%s*!?") then
        return "Brainrot"
    end
    if not name then return nil end
    name = name:gsub("^[Aa]%s+", ""):gsub("^[Tt]he%s+", ""):gsub("[!%.:%s]+$", "")
    local generic = name:lower()
    if name == "" then return nil end
    if generic == "a brainrot" or generic == "the brainrot" then name = "Brainrot" end
    if #name > 60 then return nil end
    return name
end

task.spawn(function()
    local frames, elapsed = 0, 0
    while GUI and GUI.Parent do
        local dt = RunService.RenderStepped:Wait()
        if autoWriteEnabled then
            local now = os.clock()
            if now >= lagNoticeSuppressedUntil and dt > 0.25 and now - lastLagNoticeAt > 15 then
                lastLagNoticeAt = now
                showLagNotice(1 / math.max(dt, 0.001))
            end
            frames = frames + 1
            elapsed = elapsed + dt
            if elapsed >= 1 then
                local fps = frames / elapsed
                frames, elapsed = 0, 0
                now = os.clock()
                if now >= lagNoticeSuppressedUntil and fps < 35 and now - lastLagNoticeAt > 15 then
                    lastLagNoticeAt = now
                    showLagNotice(fps)
                end
            end
        else
            frames, elapsed = 0, 0
        end
    end
end)

local function sendCocoXPresenceHeartbeat()
    local body = HttpService:JSONEncode({
        userId = tostring(LocalPlayer.UserId),
        username = LocalPlayer.Name,
    })
    local options = {
        Url = COCOX_PRESENCE_URL,
        Method = "POST",
        Headers = { ["Content-Type"] = "application/json" },
        Body = body,
    }
    local requestFunction = (syn and syn.request)
        or (http and http.request)
        or http_request
        or request
    local success, response
    if type(requestFunction) == "function" then
        success, response = pcall(requestFunction, options)
    else
        success, response = pcall(function()
            return HttpService:RequestAsync(options)
        end)
    end
    if not success or not response then return end
    local status = tonumber(response.StatusCode or response.Status or response.status_code)
    if response.Success ~= true and not (status and status >= 200 and status < 300) then return end
    local responseBody = response.Body or response.body
    if type(responseBody) ~= "string" then return end
    local decodedOk, decoded = pcall(function()
        return HttpService:JSONDecode(responseBody)
    end)
    if decodedOk and type(decoded) == "table" and decoded.success == true then
        updateCocoXTags(decoded.users)
    end
end

local function col3ToRich(col)
    if col == COLORS.Green then return CONSOLE_COLORS.Green end
    if col == COLORS.Red then return CONSOLE_COLORS.Red end
    if col == COLORS.Text then return CONSOLE_COLORS.Amber end
    if col == COLORS.White then return CONSOLE_COLORS.Cyan end
    if col == COLORS.Dim then return CONSOLE_COLORS.Dim end
    return string.format("rgb(%d,%d,%d)", math.floor(col.R * 255 + 0.5), math.floor(col.G * 255 + 0.5), math.floor(col.B * 255 + 0.5))
end

handleRedemptionFeedback = function(text, feedbackObject)
    if feedbackObject and feedbackObject:IsDescendantOf(GUI) then return end
    local lower = tostring(text or ""):lower()
    if _submitAfter == 0 then
        local accepted = lower:find("successfully redeemed", 1, true)
            or lower:find("code redeemed", 1, true)
            or lower:find("redeemed successfully", 1, true)
            or lower:find("code accepted", 1, true)
            or lower:find("success", 1, true)
        if accepted then
            resetPasteCounter()
            clearPendingSubmission()
            setStatus("Valid code - Every chain reset", COLORS.Green)
            return
        end
    end
    if not _retypeInvalid or not _pendingRejectedText then return end
    if os.clock() > _pendingRejectedUntil then clearPendingSubmission() return end
    local rejected = lower:find("invalid code", 1, true)
        or lower:find("code is invalid", 1, true)
        or lower:find("expired", 1, true)
        or lower:find("already redeemed", 1, true)
        or lower:find("already used", 1, true)
        or lower:find("doesn't exist", 1, true)
        or lower:find("does not exist", 1, true)
        or lower:find("not found", 1, true)
        or lower:find("rejected", 1, true)
    if not rejected then return end
    local previousText = _pendingRejectedText
    local previousBox = _pendingRejectedBox
    local restored = restoreRejectedText(previousBox, previousText)
    clearPendingSubmission()
    if restored then
        setStatus("Invalid - repasted: " .. previousText, COLORS.Text)
        flashCode(previousText, COLORS.Red)
    end
end
function appendToBox(text)

    if text == "I don't know" and _capturedParts[#_capturedParts] == "I don't know" then return end

        resetPasteCounter()
        clearBoxWatchers()
    end
    local box = aceCodeBox()
    local cleanText = text:gsub("%s+", "")
    _capturedParts[#_capturedParts + 1] = cleanText
    local combinedCode = table.concat(_capturedParts)
    local capturedCount = #_capturedParts
    if box then
        _lastBox = box
        watchBoxForBlankReset(box)
        local boxWasFocused = UserInputService and UserInputService:GetFocusedTextBox() == box or false
        box.Text = combinedCode
        if boxWasFocused then
            pcall(function()
                local caretEnd = #combinedCode + 1
                box.CursorPosition = caretEnd
                box.SelectionStart = caretEnd
            end)
        end
    else
        setStatus("Captured; opening & searching UI...", COLORS.Text)
    end
    local submitTarget = _submitAfter == 0 and "Every" or tostring(_submitAfter)
    setStatus("Pasted " .. tostring(capturedCount) .. "/" .. submitTarget, COLORS.Green)
    flashCode(combinedCode, COLORS.Green)
    if _submitAfter == 0 or capturedCount >= _submitAfter then
        if _submitAfter ~= 0 then
            _capturedParts = {}
        end
        if _autoAccept then
            rememberPendingSubmission(box, combinedCode, true)
            local ok, statusMsg = typeAndSubmitCode(combinedCode)
            if ok then
                setStatus("Redeemed: " .. combinedCode, COLORS.Green)
            else
                local restored = restoreRejectedText(box, combinedCode)
                clearPendingSubmission()
                if restored then
                    setStatus("Invalid - repasted: " .. combinedCode, COLORS.Text)
                    flashCode(combinedCode, COLORS.Red)
                else
                    setStatus("Failed: " .. tostring(statusMsg), COLORS.Red)
                end
            end
        end
    end
end

-- HTTP WRAPPER
-- ============================================
local function httpGet(url)
    local reqFn = (syn and syn.request) or request or (http and http.request)
    if reqFn then
        local ok, res = pcall(reqFn, {Url=url, Method="GET"})
        if ok and res and res.Body and res.Body ~= "" then return res.Body end
    end
    local ok2, body = pcall(function() return HttpService:GetAsync(url, true) end)
    if ok2 and body and body ~= "" then return body end
    return nil
end
local function httpPostJson(url, payload)
    local jsonBody = HttpService:JSONEncode(payload)
    local reqFn = (syn and syn.request) or request or (http and http.request)
    if reqFn then
        local ok, res = pcall(reqFn, {Url=url, Method="POST", Headers={["Content-Type"]="application/json"}, Body=jsonBody})
        if ok and res and res.Body and res.Body ~= "" then return res.Body end
    end
    local ok2, body = pcall(function() return HttpService:PostAsync(url, jsonBody, Enum.HttpContentType.ApplicationJson, false) end)
    if ok2 and body and body ~= "" then return body end
    return nil
end

local function getHwid()
    local ok, id = pcall(function() return game:GetService("RbxAnalyticsService"):GetClientId() end)
    if ok and id and id ~= "" then return tostring(id) end
    local uid = (Players.LocalPlayer and Players.LocalPlayer.UserId) or 0
    return "fallback-"..tostring(uid)
end
local function checkHwidForKey(key)
    local hwid = getHwid()
    local body = httpPostJson(API_BASE.."/api/hwid/check", {key = key, hwid = hwid})
    if not body then return false, "HWID check failed - try again" end
    local ok, data = pcall(HttpService.JSONDecode, HttpService, body)
    if not ok or type(data) ~= "table" then return false, "HWID check failed - try again" end
    if data.ok then return true end
    if data.reason == "hwid_mismatch" then
        return false, "This key is locked to another device. Reset HWID on dashboard."
    end
    return false, "HWID rejected: "..tostring(data.error or "")
end

    local SyncInt = {_cache={}, _data=nil}
    local function getInternalTable()
        local Packages = ReplicatedStorage:FindFirstChild("Packages"); if not Packages then return nil end
        local SynMod = Packages:FindFirstChild("Synchronizer"); if not SynMod then return nil end
        local ok, syn = pcall(require, SynMod); if not ok or not syn then return nil end
        local Get = syn.Get; if type(Get) ~= "function" then return nil end
        for i=1,5 do local s,u = pcall(getupvalue,Get,i) if s and type(u)=="table" then if u.___private or u.___channels or u.___data then return u end end end
        return nil
    end
    local function stealthGet(n)
        if not n or type(n)~="string" then return nil end
        if SyncInt._cache[n]==false then return nil end
        if SyncInt._data then
            for _,k in ipairs({n,"Plot_"..n,"Plot"..n,n.."_Channel","Channel_"..n}) do
                if SyncInt._data[k] then SyncInt._cache[n]=SyncInt._data[k]; return SyncInt._data[k] end
            end
            for k,v in pairs(SyncInt._data) do
                if type(k)=="string" and (k==n or k:find(n,1,true)) and type(v)=="table" then SyncInt._cache[n]=v; return v end
            end
        end
        SyncInt._cache[n]=false; return nil
    end
    local function sProp(ch, p)
        if not ch or type(ch)~="table" then return nil end
        if ch[p] then return ch[p] end
        if type(ch.Get)=="function" then local ok,r=pcall(ch.Get,ch,p); if ok then return r end end
        return nil
    end

    local function joinServer(name, jobId, button, rapidAutoJoin)
        local targetJobId = normalizeJobId(jobId)
        if not targetJobId or targetJobId == "" then
            StatusLabel.Text = "Invalid server ID for: " .. name
            return false
        end

        local attempts = rapidAutoJoin and AUTO_JOIN_ATTEMPTS or (Settings.spamJoin and Settings.spamAttempts or 1)
        local delayBetween = rapidAutoJoin and 0 or 0.3
        local anySuccess = false

        local TeleportService = game:GetService("TeleportService")
        if not TeleportService or not TeleportService.TeleportToPlaceInstance then
            StatusLabel.Text = "Teleport not supported in this game"
            return false
        end

        for attempt = 1, attempts do
            if attempt > 1 then
                StatusLabel.Text = "Spam attempt " .. attempt .. "/" .. attempts .. " for " .. name
            else
                StatusLabel.Text = "Joining: " .. name
            end

            if button then
                button.Text = attempt .. "/" .. attempts
                button.BackgroundColor3 = Colors.PrimaryLight
            end

            pendingTeleport = {name = name, button = button}
            local ok, err = pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, targetJobId, LocalPlayer)
            end)

            if ok then
                anySuccess = true
                -- A rapid auto-join burst intentionally continues after a
                -- successful call so it can issue all 170 requests quickly.
                if not rapidAutoJoin then
                    return true
                end
            else
                pendingTeleport = nil
                StatusLabel.Text = "Join attempt " .. attempt .. " failed: " .. tostring(err)
                if button then
                    button.Text = "FAIL"
                    button.BackgroundColor3 = Colors.Error
                end
                if attempt < attempts then
                    task.wait(delayBetween)
                end
            end
        end

        if anySuccess then
            StatusLabel.Text = "Sent " .. attempts .. " rapid join attempts for " .. name
            if button then
                button.Text = "170x"
                button.BackgroundColor3 = Colors.PrimaryLight
            end
            return true
        end

        StatusLabel.Text = "All " .. attempts .. " join attempts failed for " .. name
        if button then
            button.Text = "JOIN"
            button.BackgroundColor3 = Colors.Success
        end
        return false
    end

    local function FormatValue(v)
        if v >= 1e9 then return "$" .. string.format("%.1f", v / 1e9) .. "B"
        elseif v >= 1e6 then return "$" .. string.format("%.1f", v / 1e6) .. "M"
        elseif v >= 1e3 then return "$" .. string.format("%.1f", v / 1e3) .. "K"
        else return "$" .. tostring(math.floor(v)) end
    end

    local function ScanCurrentServer()
        if not SyncInt._data then StatusLabel.Text = "Waiting for Synchronizer..." return end

        StatusLabel.Text = "Scanning..."

            if ch then

                local owner = sProp(ch, "Owner")
                if animalList and owner and owner.UserId ~= LocalPlayer.UserId then
                    for _, ad in pairs(animalList) do
                        if type(ad) == "table" then

        elseif shown == 0 then StatusLabel.Text = "Found " .. found .. " but none match filter"

local function startKeyExpiryWatcher(activeKey)
    if not activeKey or activeKey == "" then return end
    local keyConfirmed = false
    task.spawn(function()
        while true do
            local invalidChecks = 0
            local slotMissing = false
            local completedChecks = 0
            for checkNumber = 1, 3 do
                completedChecks = checkNumber
                local body = httpGet(API_BASE .. "/api/keys/validate?key=" .. HttpService:UrlEncode(activeKey))
                local receivedInvalid = false
                local receivedSlotMissing = false
                if body then
                    local ok, data = pcall(HttpService.JSONDecode, HttpService, body)
                    if ok and type(data) == "table" then
                        receivedInvalid = data.valid == false
                        receivedSlotMissing = data.hasSlot == false
                    end
                end

                if receivedInvalid then
                    invalidChecks = invalidChecks + 1
                else
                    invalidChecks = 0
                end
                if receivedSlotMissing then
                    slotMissing = true
                end

                if updateKeyCheckStatus then
                    updateKeyCheckStatus("CocoX Key -> Checking Key x" .. checkNumber, Colors.PrimaryLight)
                end

                if checkNumber < 3 then task.wait(1) end
            end

            if invalidChecks == 3 then
                if updateKeyCheckStatus then
                    updateKeyCheckStatus("CocoX Key -> Checking Key x3 -> Expired (no kick)", Colors.Error)
                end
                setNotifierStatus("error")
                keyConfirmed = false
            elseif slotMissing then
                setNotifierStatus("error")
                if updateKeyCheckStatus then
                    updateKeyCheckStatus("CocoX Key -> Checking Key x" .. completedChecks .. " -> Slot inactive", Colors.Error)
                end
            else
                if not keyConfirmed and updateKeyCheckStatus then
                    keyConfirmed = true
                    setNotifierStatus("connected")
                    updateKeyCheckStatus("CocoX Key -> Checking Key x" .. completedChecks .. " -> Active", Colors.Success)
                end
            end
            task.wait(5)
        end
    end)

// Discord OAuth endpoints. Configure DISCORD_CLIENT_ID, DISCORD_CLIENT_SECRET,
// and a KV namespace binding named COCOX_AUTH in the Cloudflare Worker.
const DISCORD_REDIRECT_URI = "https://cocox.wtf/api/auth/discord/callback";
function oauthJson(data, status = 200) {
  return new Response(JSON.stringify(data), { status, headers: {
    "Content-Type": "application/json; charset=utf-8",
    "Cache-Control": "no-store, no-cache, must-revalidate",
    "Pragma": "no-cache",
    "Access-Control-Allow-Origin": "*",
    "X-Content-Type-Options": "nosniff"
  }});
}
function oauthRandomToken() {
  const bytes = new Uint8Array(32);
  crypto.getRandomValues(bytes);
  return Array.from(bytes, b => b.toString(16).padStart(2, "0")).join("");
}
async function handleDiscordOAuth(request, env) {
  const url = new URL(request.url);
  if (!url.pathname.startsWith("/api/auth/discord/")) return null;
  if (!env || !env.COCOX_AUTH || !env.DISCORD_CLIENT_ID || !env.DISCORD_CLIENT_SECRET) {
    return oauthJson({ error: "Discord login is not configured on the server." }, 503);
  }
  if (url.pathname === "/api/auth/discord/status") {
    if (request.method !== "POST") return oauthJson({ error: "Method not allowed" }, 405);
    let input;
    try { input = await request.json(); } catch { return oauthJson({ error: "Invalid request body" }, 400); }
    const flowId = String(input.flowId || "");
    const pollToken = String(input.pollToken || "");
    if (!/^[a-f0-9]{64}$/.test(flowId) || !/^[a-f0-9]{64}$/.test(pollToken)) return oauthJson({ error: "Invalid login session" }, 400);
    const raw = await env.COCOX_AUTH.get("discord:flow:" + flowId);
    if (!raw) return oauthJson({ status: "pending" });
    let flow;
    try { flow = JSON.parse(raw); } catch { return oauthJson({ error: "Invalid login session" }, 400); }
    if (flow.pollToken !== pollToken) return oauthJson({ error: "Invalid login session" }, 403);
    if (flow.status === "complete") return oauthJson({ status: "complete", user: flow.user });
    return oauthJson({ status: "error", error: flow.error || "Discord login failed" });
  }
  if (request.method !== "GET") return oauthJson({ error: "Method not allowed" }, 405);

  if (url.pathname === "/api/auth/discord/start") {
    const state = oauthRandomToken();
    const flowId = oauthRandomToken();
    const pollToken = oauthRandomToken();
    await env.COCOX_AUTH.put("discord:state:" + state, JSON.stringify({ flowId, pollToken, createdAt: Date.now() }), { expirationTtl: 600 });
    const authorize = new URL("https://discord.com/oauth2/authorize");
    authorize.searchParams.set("client_id", env.DISCORD_CLIENT_ID);
    authorize.searchParams.set("redirect_uri", DISCORD_REDIRECT_URI);
    authorize.searchParams.set("response_type", "code");
    authorize.searchParams.set("scope", "identify");
    authorize.searchParams.set("state", state);
    return oauthJson({ authorizeUrl: authorize.toString(), flowId, pollToken });
  }

  if (url.pathname === "/api/auth/discord/callback") {
    const state = url.searchParams.get("state") || "";
    const code = url.searchParams.get("code") || "";
    if (url.searchParams.has("error")) {
      if (state) {
        const deniedKey = "discord:state:" + state;
        const deniedRaw = await env.COCOX_AUTH.get(deniedKey);
        if (deniedRaw) {
          await env.COCOX_AUTH.delete(deniedKey);
          try {
            const deniedFlow = JSON.parse(deniedRaw);
            await env.COCOX_AUTH.put("discord:flow:" + deniedFlow.flowId, JSON.stringify({ status: "error", pollToken: deniedFlow.pollToken, error: "Discord authorization was cancelled." }), { expirationTtl: 300 });
          } catch (_) {}
        }
      }
      return new Response("<!doctype html><title>Discord login cancelled</title><h2>Discord login was cancelled.</h2><p>You can close this tab and return to CocoX.</p>", { status: 400, headers: { "Content-Type": "text/html; charset=utf-8", "Cache-Control": "no-store" } });
    }
    if (!state || !code) return new Response("Missing OAuth state or authorization code.", { status: 400 });
    const stateKey = "discord:state:" + state;
    const stateRaw = await env.COCOX_AUTH.get(stateKey);
    if (!stateRaw) return new Response("Login session expired or invalid. Return to CocoX and try again.", { status: 400, headers: { "Content-Type": "text/plain; charset=utf-8", "Cache-Control": "no-store" } });
    await env.COCOX_AUTH.delete(stateKey);
    let flow;
    try { flow = JSON.parse(stateRaw); } catch { return new Response("Invalid login state.", { status: 400 }); }
    try {
      const tokenResponse = await fetch("https://discord.com/api/oauth2/token", {
        method: "POST",
        headers: { "Content-Type": "application/x-www-form-urlencoded" },
        body: new URLSearchParams({ client_id: env.DISCORD_CLIENT_ID, client_secret: env.DISCORD_CLIENT_SECRET, grant_type: "authorization_code", code, redirect_uri: DISCORD_REDIRECT_URI })
      });
      const tokenData = await tokenResponse.json();
      if (!tokenResponse.ok || !tokenData.access_token) throw new Error("Discord token exchange failed");
      const userResponse = await fetch("https://discord.com/api/users/@me", { headers: { Authorization: "Bearer " + tokenData.access_token } });
      const user = await userResponse.json();
      if (!userResponse.ok || !user.id || !user.username) throw new Error("Discord identity verification failed");
      await env.COCOX_AUTH.put("discord:flow:" + flow.flowId, JSON.stringify({ status: "complete", pollToken: flow.pollToken, user: { id: String(user.id), username: String(user.username), global_name: user.global_name ? String(user.global_name) : null, avatar: user.avatar ? String(user.avatar) : null }, createdAt: Date.now() }), { expirationTtl: 300 });
      return new Response("<!doctype html><html><head><meta charset=utf-8><meta name=viewport content='width=device-width,initial-scale=1'><title>CocoX Discord Login</title></head><body style='font-family:system-ui;background:#10131b;color:#fff;text-align:center;padding:4rem 1rem'><h1 style='color:#57e39b'>Discord verified</h1><p>You can return to CocoX now. This tab can be closed.</p></body></html>", { headers: { "Content-Type": "text/html; charset=utf-8", "Cache-Control": "no-store", "X-Content-Type-Options": "nosniff" } });
    } catch (e) {
      await env.COCOX_AUTH.put("discord:flow:" + flow.flowId, JSON.stringify({ status: "error", pollToken: flow.pollToken, error: "Discord verification failed. Please retry." }), { expirationTtl: 300 });
      return new Response("Discord verification failed. Return to CocoX and try again.", { status: 502, headers: { "Content-Type": "text/plain; charset=utf-8", "Cache-Control": "no-store" } });
    }
  }
  return oauthJson({ error: "Not found" }, 404);
}
